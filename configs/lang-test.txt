active zhongli;

let combos = 0;
fn hutao_combo(n) {
  let i = 0;
  while 1 {
    if i >= n { break; }
    hutao attack:2, charge, dash;
    i = i + 1;
  }
  return n * 2;
}
fn pick(x) {
  switch x {
  case 0:
    return 3;
  case 1:
    fallthrough;
  case 2:
    return 4;
  default:
    return 5;
  }
}
let rounds = 0;
while 1 {
  zhongli skill[hold=1];
  yelan skill;
  if .yelan.burst.ready && .yelan.energy >= .yelan.energymax {
    yelan burst, attack;
  }
  if .xingqiu.burst.ready {
    xingqiu burst;
  } else if .xingqiu.skill.cd > 120 {
    wait(10);
  } else {
    delay(5);
  }
  for let k = 0; k < 3; k = k + 1 {
    if .xingqiu.skill.ready { break; }
    xingqiu attack;
  }
  while !.xingqiu.skill.ready { xingqiu attack; }
  xingqiu skill;
  if .xingqiu.skill.ready { xingqiu skill; }
  execute_action(.keys.char.hutao, .action.skill, []);
  hutao attack, charge, dash;
  combos = combos + hutao_combo(pick(rounds - (rounds / 3) * 3) + 3);
  if .hutao.burst.ready && .hutao.hpratio <= 0.5 || f() > 100000 {
    hutao burst;
  }
  rounds = rounds + 1;
}
